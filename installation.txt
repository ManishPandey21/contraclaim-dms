def clean_text(text):
    """Removes tags and extra whitespace."""
    text = re.sub(r'\', '', text)
    text = re.sub(r'--- PAGE \d+ ---', '', text)
    return text.strip()

def extract_content(text):
    """Extracts Subject, Body, and determines Persona."""
    
    # 1. Determine Persona (System Prompt)
    if "TYPSA" in text or "ITALFERR" in text or "GCS" in text:
        role = "system"
        persona = (
            "You are the Project Director for the General Consultant (Typsa-Italferr JV). "
            "You enforce contract compliance strictly. Write formal notices involving penalties, "
            "safety violations, and claim rejections, citing specific contract clauses. "
            "Tone: Authoritative, punitive, and direct."
        )
    elif "AFCONS" in text or "SAM INDIA" in text or "GULERMAK" in text:
        role = "system"
        persona = (
            "You are the Project Manager for the Contractor (Afcons-Sam India / Gulermak). "
            "You write letters regarding additional costs, variations, and site handovers. "
            "Always reserve your rights to add or amend submissions. "
            "Tone: Professional, assertive, and documentation-focused."
        )
    else:
        persona = "You are a Contract Manager drafting formal correspondence."

    # 2. Extract Subject (User Prompt)
    subject_match = re.search(r'Subject:\s*(.*?)(?=\n|Ref\.:|Dear)', text, re.DOTALL | re.IGNORECASE)
    if subject_match:
        subject_text = clean_text(subject_match.group(1)).replace('\n', ' ')
        user_prompt = f"Draft a formal letter regarding: {subject_text}"
    else:
        user_prompt = "Draft a formal contract letter regarding the current project issues."

    # 3. Extract Body (Assistant Response)
    # Tries to find text between "Dear Sir" and the sign-off
    start_match = re.search(r'Dear Sir,', text, re.IGNORECASE)
    end_match = re.search(r'(Thanking you|Yours faithfully|Yours Truly)', text, re.IGNORECASE)

    if start_match and end_match:
        start_idx = start_match.end()
        end_idx = end_match.start()
        body_text = clean_text(text[start_idx:end_idx])
    else:
        # Fallback if structure is messy
        body_text = clean_text(text[-1000:]) # Take last 1000 chars

    return persona, user_prompt, body_text

def clean_text(text):
    """Removes , page numbers, and excessive whitespace."""
    # Remove source tags like 
    text = re.sub(r'\', '', text)
    # Remove page markers
    text = re.sub(r'--- PAGE \d+ ---', '', text)
    # Remove multiple newlines and spaces
    text = re.sub(r'\s+', ' ', text).strip()
    return text

def identify_persona(text):
    """Determines if the sender is the Authority (GC) or Contractor."""
    text_start = text[:500].upper() # Check header area
    
    if "TYPSA" in text_start or "ITALFERR" in text_start:
        return {
            "role": "General Consultant (Authority)",
            "system_prompt": "You are the Project Director for the General Consultant (Typsa-Italferr JV). You enforce contract compliance strictly. Write formal notices, reject inadmissible claims, and cite contract clauses (GCC, SHE) authoritatively."
        }
    elif "AFCONS" in text_start or "SAM INDIA" in text_start or "GULERMAK" in text_start:
        return {
            "role": "Contractor (Claimant)",
            "system_prompt": "You are the Project Manager for the Contractor (Afcons-Sam India / Gulermak-Sam India). You write professional letters regarding site issues, reimbursements, and approvals. Always justify your claims using facts and reserve your rights to amend submissions."
        }
    else:
        return {
            "role": "Unknown",
            "system_prompt": "You are a professional Contract Manager writing formal correspondence."
        }

def extract_subject(text):
    """Attempts to extract the subject line to create a synthetic user prompt."""
    match = re.search(r'Subject:\s*(.*?)(?=\s*(Ref\.|Reference:|Dear|Att\.))', text, re.IGNORECASE)
    if match:
        return match.group(1).strip()
    return "Contractual Correspondence"

def generate_jsonl(data_list, output_filename="contract_finetune_data.jsonl"):
    output_data = []
    
    for entry in data_list:
        raw_text = entry['content']
        cleaned_body = clean_text(raw_text)
        
        # 1. Identify Persona
        persona_info = identify_persona(cleaned_body)
        
        # 2. Extract Subject for the "User" Prompt
        subject = extract_subject(cleaned_body)
        
        # 3. Create the Synthetic User Instruction
        # This simulates what a human would ask the AI to generate this specific letter.
        if "General Consultant" in persona_info['role']:
            user_content = f"Draft a formal letter regarding: '{subject}'. Reference relevant contract clauses and take a firm stance on compliance or admissibility."
        else:
            user_content = f"Draft a formal letter regarding: '{subject}'. Clearly state the facts, requested actions (like reimbursement or approval), and ensure we reserve our rights."

        # 4. Construct the JSON Object
        json_entry = {
            "messages": [
                {"role": "system", "content": persona_info['system_prompt']},
                {"role": "user", "content": user_content},
                {"role": "assistant", "content": cleaned_body}
            ]
        }
        output_data.append(json_entry)

    # 5. Save to File
    with open(output_filename, 'w', encoding='utf-8') as f:

            f.write(json.dumps(item) + "\n")
